sakuya_superdip_humiliate_01 = {
	groups = {
		general
	}

	texture = "gfx/interface/sakuya/crino.dds"
	requires_approval = no
	should_notify_third_parties = no
	show_in_lens = no
	show_effect_in_tooltip = yes
	show_confirmation_box = yes

	selectable = {
		is_player = yes
		has_variable = SKY_RIGHT_CLICK_ON
		NOT = { is_country_type = decentralized }
	}

	potential = {
		has_variable = SKY_RIGHT_CLICK_ON
		scope:target_country ?= {
			is_player = no
			NOT = { is_country_type = decentralized }
		}
	}

	possible = {
		root != scope:target_country
		NOT = { has_war_with = scope:target_country }
		scope:target_country ?= {
			NOT = {
				has_diplomatic_pact = {
					who = root
					type = humiliation
				}
			}
			NOT = {
				has_diplomatic_pact = {
					who = root
					type = sakuya_superdip_humiliation_pact_01
				}
			}
		}
	}

	accept_effect = {
		save_scope_as = sakuya_superdip_actor_country
		set_variable = { name = sakuya_superdip_target_country_var value = scope:target_country }
		trigger_event = {
			id = sakuya_superdip.1
			popup = yes
		}
	}

	ai = {
		will_propose = {
			always = no
		}
	}

	is_hostile = yes
}

sakuya_superdip_subjugate_country_01 = {
	groups = {
		general
	}

	texture = "gfx/interface/sakuya/crino.dds"
	requires_approval = no
	should_notify_third_parties = no
	show_in_lens = no
	show_effect_in_tooltip = yes
	show_confirmation_box = yes

	selectable = {
		is_player = yes
		has_variable = SKY_RIGHT_CLICK_ON
	}

	potential = {
		has_variable = SKY_RIGHT_CLICK_ON
		scope:target_country ?= {
			is_player = no
			NOT = { is_country_type = decentralized }
			is_subject = no
		}
	}

	possible = {
		root != scope:target_country
		scope:target_country = {
			is_diplomatic_play_committed_participant = no
		}
	}

	accept_effect = {
		save_scope_as = actor
		scope:target_country = {
			save_scope_as = target
		}
		if = {
			limit = { is_country_type = unrecognized }
			create_diplomatic_pact = {
				country = scope:target_country
				type = tributary
			}
		}
		else_if = {
			limit = {
				scope:target_country = {
					is_country_type = colonial
				}
			}
			create_diplomatic_pact = {
				country = scope:target_country
				type = dominion
			}
		}
		else = {
			create_diplomatic_pact = {
				country = scope:target_country
				type = protectorate
			}
		}
		post_notification = sakuya_superdip_subjugate_country_toast
	}

	ai = {
		will_propose = {
			always = no
		}
	}

	is_hostile = yes
}
