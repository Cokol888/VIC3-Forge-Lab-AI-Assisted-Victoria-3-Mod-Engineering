sakuya_b5_apply_sel_approval_delta_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = sakuya_b5_sel_approval_mult
			}
		}
		set_variable = {
			name = sakuya_b5_sel_approval_mult
			value = 0
		}
	}
	change_variable = {
		name = sakuya_b5_sel_approval_mult
		add = $DELTA$
	}
	if = {
		limit = {
			var:sakuya_b5_sel_approval_mult = 0
		}
		remove_variable = sakuya_b5_sel_approval_mult
	}
	sakuya_b5_apply_sel_ig_modifier_effect = {
		MOD = sakuya_cheat_b5_sel_approval
		VAR = sakuya_b5_sel_approval_mult
	}
	sakuya_b5_refresh_selected_approval_active_effect = yes
}
