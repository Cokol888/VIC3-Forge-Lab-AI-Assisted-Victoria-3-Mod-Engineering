sakuya_b3_amendment_add_to_law_auto_sponsor_effect = {
	root = {
		random_interest_group = {
			save_scope_as = sakuya_b3_amendment_sponsor
		}
	}
	if = {
		limit = { exists = scope:sakuya_b3_amendment_sponsor }
		add_amendment = {
			type = $TYPE$
			sponsor = scope:sakuya_b3_amendment_sponsor
			cooldown = 120
		}
	}
}

sakuya_b3_amendment_add_to_law_selected_sponsor_effect = {
	root = { sakuya_b3_amendment_save_selected_ig_effect = yes }
	if = {
		limit = {
			NOT = { exists = scope:sakuya_b3_amendment_sponsor }
		}
		root = {
			random_interest_group = {
				save_scope_as = sakuya_b3_amendment_sponsor
			}
		}
	}
	if = {
		limit = { exists = scope:sakuya_b3_amendment_sponsor }
		add_amendment = {
			type = $TYPE$
			sponsor = scope:sakuya_b3_amendment_sponsor
			cooldown = 120
		}
	}
}
