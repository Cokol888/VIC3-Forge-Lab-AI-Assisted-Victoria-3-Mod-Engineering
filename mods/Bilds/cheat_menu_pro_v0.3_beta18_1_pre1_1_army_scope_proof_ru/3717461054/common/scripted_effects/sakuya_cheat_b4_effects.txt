    every_country = {

    }
}

# 解析目标州（用于 state-based wargoals）
# 正向（玩家攻目标）：目标国家有标记州(sakuya_mark_foreign_state_target) → 用它，否则用首都
# 反向（目标攻玩家）：玩家有标记州(decree_sakuya_mark_state) → 用它，否则用首都
# 结果保存为 scope:sakuya_dp_target_state
sakuya_b4_resolve_marked_political_movement_effect = {
    remove_variable = sakuya_b4_marked_political_movement_found
    every_political_movement = {
        limit = { has_variable = sakuya_main_04_03_marked_political_movement }
        save_scope_as = sakuya_b4_marked_political_movement_scope
        root = { set_variable = sakuya_b4_marked_political_movement_found }
    }
    if = {
        limit = {
            NOT = { has_variable = sakuya_b4_marked_political_movement_found }
        }
        every_country = {
            limit = {
                any_political_movement = { has_variable = sakuya_main_04_03_marked_political_movement }
            }
            every_political_movement = {
                limit = { has_variable = sakuya_main_04_03_marked_political_movement }
                save_scope_as = sakuya_b4_marked_political_movement_scope
                root = { set_variable = sakuya_b4_marked_political_movement_found }
            }
        }
    }
}

sakuya_b4_resolve_target_state_effect = {
    # 默认：使用目标国家的首都
    scope:sakuya_dp_target = {
        capital = { save_scope_as = sakuya_dp_target_state }
    }
    # 如果 ROOT == 玩家（正向），在目标国家找外国州标记
    if = {
        limit = { is_player = yes }
        scope:sakuya_dp_target = {
            random_scope_state = {
                limit = { has_variable = sakuya_mark_foreign_state_target }
                save_scope_as = sakuya_dp_target_state
            }
        }
    }
    # 如果 ROOT != 玩家（反向，目标攻玩家），在玩家（=sakuya_dp_target）找法令标记
    else = {
        scope:sakuya_dp_target = {
            random_scope_state = {
                limit = { has_decree = decree_sakuya_mark_state }
                save_scope_as = sakuya_dp_target_state
            }
        }
    }
}

        create_diplomatic_play = { type = dp_conquer_state target_state = scope:sakuya_dp_target_state }
    }
    else_if = {
        limit = { scope:sakuya_dp_player = { has_variable = sakuya_b4_wg_return_state } }

        limit = { scope:sakuya_dp_player = { has_variable = sakuya_b4_wg_take_treaty_port } }
