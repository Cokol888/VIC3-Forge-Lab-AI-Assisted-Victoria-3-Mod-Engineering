cmp_target_core_toggle_group_a_effect = {
    remove_variable = cmp_target_core_group_op_found
    every_country = {
        limit = { has_variable = sakuya_mark_country_target }
        if = {
            limit = { has_variable = cmp_target_group_a_member }
            remove_variable = cmp_target_group_a_member
            root = { set_variable = { name = cmp_target_core_feedback value = 3 } }
        }
        else = {
            set_variable = cmp_target_group_a_member
            root = { set_variable = { name = cmp_target_core_feedback value = 2 } }
        }
        root = { set_variable = cmp_target_core_group_op_found }
    }
    if = { limit = { NOT = { has_variable = cmp_target_core_group_op_found } } set_variable = { name = cmp_target_core_feedback value = -1 } }
    remove_variable = cmp_target_core_group_op_found
}

# Clears transient CMP marks. Own-state decree marks and A/B/C persistent groups are deliberately retained.
cmp_target_core_clear_transient_marks_effect = {
    every_country = { limit = { has_variable = sakuya_mark_country_target } remove_variable = sakuya_mark_country_target }
    ordered_decentralized_country = {
        limit = { has_variable = sakuya_mark_country_target }
        order_by = gdp
        position = 0
        check_range_bounds = no
        remove_variable = sakuya_mark_country_target
    }
    every_state = { limit = { has_variable = sakuya_mark_foreign_state_target } remove_variable = sakuya_mark_foreign_state_target }
    every_country = {
        every_scope_character = { limit = { has_variable = sakuya_mark_character_target } remove_variable = sakuya_mark_character_target }
        if = { limit = { has_variable = sakuya_mark_power_bloc_target } remove_variable = sakuya_mark_power_bloc_target }
        if = { limit = { has_variable_list = sakuya_main_04_01_marked_army_list } clear_variable_list = sakuya_main_04_01_marked_army_list }
        if = { limit = { has_variable_list = sakuya_main_04_01_marked_navy_list } clear_variable_list = sakuya_main_04_01_marked_navy_list }
        every_military_formation = {
            if = { limit = { has_variable = sakuya_main_04_01_marked_navy } remove_variable = sakuya_main_04_01_marked_navy }
            if = { limit = { has_variable = cmp_military_target_fleet } remove_variable = cmp_military_target_fleet }
        }
    }
    every_character_in_exile_pool = { limit = { has_variable = sakuya_mark_character_target } remove_variable = sakuya_mark_character_target }
    if = {
        limit = { OR = { var:cmp_target_core_mode = 2 var:cmp_target_core_mode = 8 var:cmp_target_core_mode = 9 var:cmp_target_core_mode = 10 var:cmp_target_core_mode = 11 } }
        set_variable = { name = cmp_target_core_mode value = 1 }
    }
    set_variable = { name = cmp_target_core_feedback value = 5 }
}
