# === 按钮1 行政力 ===
sakuya_b1_bureaucracy_check = {
	scope = country
	is_shown = {
		OR = {
			has_variable = sakuya_bureaucracy_active
			has_modifier = sakuya_cheat_b1_bureaucracy_percent
			any_country = { has_variable = sakuya_mark_country_target has_modifier = sakuya_cheat_b1_bureaucracy_percent_marked_plus25 }
			any_country = { has_variable = sakuya_mark_country_target has_modifier = sakuya_cheat_b1_bureaucracy_percent_marked_minus25 }
		}
	}
} 
 
sakuya_b1_bureaucracy_add = {
	scope = country
	is_shown = { is_player = yes }
	effect = {
		if = {
			limit = { NOT = { has_variable = sakuya_bureaucracy_mult } }
			set_variable = { name = sakuya_bureaucracy_mult value = 0 }
		}
		change_variable = { name = sakuya_bureaucracy_mult add = 500 }
		remove_modifier = sakuya_cheat_bureaucracy
		add_modifier = { name = sakuya_cheat_bureaucracy multiplier = var:sakuya_bureaucracy_mult }
		set_variable = sakuya_bureaucracy_active
	}
}
