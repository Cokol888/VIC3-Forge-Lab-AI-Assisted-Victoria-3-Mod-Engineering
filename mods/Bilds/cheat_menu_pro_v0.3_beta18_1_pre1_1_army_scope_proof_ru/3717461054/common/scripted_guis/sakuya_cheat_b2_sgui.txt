# --- 1. Government Dividends Efficiency ---
sakuya_b2_tax_income_check = {
	scope = country
	is_shown = { has_variable = sakuya_b2_tax_income_active }
}

sakuya_b2_tax_income_add = {
	scope = country
	is_shown = { is_player = yes }
	effect = {
		add_modifier = { name = sakuya_cheat_b2_tax_income multiplier = 0.2 }
		set_variable = sakuya_b2_tax_income_active
	}
}

sakuya_b2_tax_income_add_shift = {

	is_shown = { is_player = yes }

	}

	}

    scope = country

    scope = country
    is_shown = { is_player = yes }
    effect = {
        if = {
            limit = { NOT = { has_variable = sakuya_b2_minting_mult } }
            set_variable = { name = sakuya_b2_minting_mult value = 0 }

        change_variable = { name = sakuya_b2_minting_mult add = 0.2 }

        add_modifier = { name = sakuya_cheat_b2_minting multiplier = var:sakuya_b2_minting_mult }
